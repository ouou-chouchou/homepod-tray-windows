cmake_minimum_required(VERSION 3.20)
project(HomePodTray LANGUAGES C CXX RC)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(AIRPLAY_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(AIRPLAY_BUILD_TESTS ON CACHE BOOL "" FORCE)
add_subdirectory(vendor/airplay2-sender-cpp)
target_include_directories(raop_sender PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/app)
if(MSVC)
    target_compile_options(raop_sender PRIVATE /utf-8)
    target_compile_options(airplay_crypto PRIVATE /utf-8)
endif()
add_executable(HomePodTray WIN32 app/tray.cpp app/audio.cpp app/discovery.cpp app/resources.rc)
target_link_libraries(HomePodTray PRIVATE raop_sender ole32 uuid mmdevapi shell32 advapi32 comctl32 iphlpapi)
target_compile_definitions(HomePodTray PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(HomePodTray PRIVATE /utf-8 /W4)
enable_testing()
add_executable(audio_policy_tests tests/audio_policy_tests.cpp)
target_include_directories(audio_policy_tests PRIVATE app)
target_compile_definitions(audio_policy_tests PRIVATE NOMINMAX)
add_test(NAME audio_policy_tests COMMAND audio_policy_tests)
